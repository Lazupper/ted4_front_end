let nomes = ["Ana", "Bruno", "Carlos", "Diana", "Eduardo"];
console.log("Array inicial de nomes:", nomes);
console.log("Terceiro nome (índice 2):", nomes[2]);

nomes.push("Fábio");    
nomes.unshift("Gabriela"); 
console.log("Array após adições:", nomes);

nomes.pop(); 
console.log("Array após remover o último nome (pop):", nomes);

let numeros = [2, 4, 6, 8];
let numerosDobrados = numeros.map(numero => numero * 2);

console.log("\nArray original para map():", numeros);
console.log("Array com valores dobrados (map):", numerosDobrados);

let valores = [1, 3, 5, 7, 9];
let maioresQueCinco = valores.filter(valor => valor > 5);

console.log("\nArray original para filter():", valores);
console.log("Array apenas com > 5 (filter):", maioresQueCinco);
